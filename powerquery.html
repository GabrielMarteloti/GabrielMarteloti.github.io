Código

consulta.m
let
    Fonte = Excel.Workbook(File.Contents("C:\\dados\\base.xlsx"), null, true),
    Tabela = Fonte{[Item="Vendas", Kind="Table"]}[Data],
    Tipos = Table.TransformColumnTypes(Tabela, {{"Data", type date}, {"Valor", type number}}),
    SemVazios = Table.SelectRows(Tipos, each [Valor] <> null)
in
    SemVazios
