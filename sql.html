Código

consulta.sql
SELECT
    c.regiao,
    COUNT(DISTINCT p.id_pedido) AS pedidos,
    SUM(p.valor)                AS faturamento
FROM pedidos p
JOIN clientes c ON c.id_cliente = p.id_cliente
WHERE p.data_pedido >= '2026-01-01'
GROUP BY c.regiao
ORDER BY faturamento DESC;
